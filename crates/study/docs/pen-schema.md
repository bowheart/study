# .pen schema (version 2.17)

A `.pen` file is JSON: `{ version, themes?, variables?, imports?, children: Node[] }`.
Only the types and properties below exist. Anything else is ignored (and reported as a warning).

```typescript
/** "$name" binds a property to a variable. */
type Var = string;
type Num = number | Var;
/** "#RGB", "#RRGGBB" or "#RRGGBBAA" */
type Color = string | Var;
type Bool = boolean | Var;

/** Sizes: a number, a variable, or
 *  "fit_content"    — size of the children (frames with layout) or of the text
 *  "fill_container" — share of the parent's space (parent must have flex layout)
 *  Optional fallback in parentheses, e.g. "fit_content(240)" — used when the size can't be derived. */
type Size = Num | string;

type BlendMode = "normal" | "darken" | "multiply" | "linearBurn" | "colorBurn" | "light" | "screen" | "linearDodge"
  | "colorDodge" | "overlay" | "softLight" | "hardLight" | "difference" | "exclusion" | "hue" | "saturation" | "color" | "luminosity";

type Fill =
  | Color
  | { type: "color"; color: Color; enabled?: Bool; blendMode?: BlendMode }
  | {
      type: "gradient"; gradientType?: "linear" | "radial" | "angular"; enabled?: Bool; blendMode?: BlendMode; opacity?: Num;
      /** Normalized to the node's box. Default {x: 0.5, y: 0.5}. */
      center?: { x?: Num; y?: Num };
      /** Normalized. Linear: height = gradient length. Radial/angular: ellipse diameters. Default 1×1. */
      size?: { width?: Num; height?: Num };
      /** Degrees, counter-clockwise. At 0 the gradient runs top (position 0) to bottom (position 1);
       *  90 runs left → right, 180 bottom → top, 270 right → left. */
      rotation?: Num;
      colors?: { color: Color; position: Num }[];
    }
  /** url is relative to the .pen file. mode: "fill" (cover, default), "fit" (contain), "stretch". */
  | { type: "image"; url: string; mode?: "stretch" | "fill" | "fit"; opacity?: Num; enabled?: Bool; blendMode?: BlendMode }
  /** WebGL 1 fragment shader file (relative url). Uniforms annotated `@resolution` / `@time` are bound
   *  automatically; others take values from `uniforms` (number, bool, "#hex" color, number array, or "$var"). */
  | { type: "shader"; url: string; uniforms?: Record<string, number | boolean | string | number[]>; opacity?: Num; enabled?: Bool; blendMode?: BlendMode }
  | { type: "mesh_gradient"; columns?: number; rows?: number; colors?: Color[]; points?: [number, number][]; opacity?: Num; enabled?: Bool };
type Fills = Fill | Fill[];   // arrays paint bottom to top

interface Stroke {
  stroke?: Fills;
  strokeWidth?: Num | { top?: Num; right?: Num; bottom?: Num; left?: Num };
  strokeAlignment?: "inner" | "center" | "outer";   // strokes never affect layout
  strokeLinecap?: "butt" | "round" | "square";
  strokeLinejoin?: "miter" | "bevel" | "round";
}

type Effect =
  | { type: "blur"; radius?: Num; enabled?: Bool }               // blurs the node
  | { type: "background_blur"; radius?: Num; enabled?: Bool }    // blurs what's behind the node
  | { type: "shadow"; shadowType?: "outer" | "inner"; offset?: { x: Num; y: Num }; blur?: Num; spread?: Num; color?: Color; blendMode?: BlendMode; enabled?: Bool };

interface Base {
  id: string;            // generated by study; never contains "/"
  name?: string;
  x?: number; y?: number;   // relative to the parent's top-left; ignored inside flex layout
  enabled?: Bool;        // false hides the node (and removes it from layout)
  opacity?: Num;
  rotation?: Num;        // degrees counter-clockwise around the top-left corner
  flipX?: Bool; flipY?: Bool;
  layoutPosition?: "auto" | "absolute";   // absolute: leaves the parent's flex flow, uses x/y
  reusable?: boolean;    // makes the node a component that `ref` nodes can instance
  theme?: Record<string, string>;         // selects theme values for this subtree, e.g. { mode: "dark" }
  context?: string;
  metadata?: { type: string; [k: string]: any };
}

interface Frame extends Base, Stroke {
  type: "frame";
  width?: Size; height?: Size;            // default fit_content
  fill?: Fills;
  cornerRadius?: Num | [Num, Num, Num, Num];   // top-left, top-right, bottom-right, bottom-left
  effect?: Effect | Effect[];
  clip?: Bool;                             // clip children to the frame. Default false
  placeholder?: boolean;                   // marks a frame as work in progress
  layout?: "horizontal" | "vertical" | "none";   // default "horizontal"
  gap?: Num;
  padding?: Num | [Num, Num] | [Num, Num, Num, Num];   // all | [vertical, horizontal] | [top, right, bottom, left]
  justifyContent?: "start" | "center" | "end" | "space_between" | "space_around";   // main axis
  alignItems?: "start" | "center" | "end";                                          // cross axis
  slot?: false | string[];
  children?: Node[];
}

interface Group extends Base { type: "group"; effect?: Effect | Effect[]; children?: Node[] }

interface Rectangle extends Base, Stroke { type: "rectangle"; width?: Size; height?: Size; fill?: Fills; cornerRadius?: Num | [Num, Num, Num, Num]; effect?: Effect | Effect[] }

interface Ellipse extends Base, Stroke {
  type: "ellipse"; width?: Size; height?: Size; fill?: Fills; effect?: Effect | Effect[];
  innerRadius?: Num;   // 0 = solid, 0..1 = ring (donut) thickness ratio
  startAngle?: Num;    // degrees CCW from the +x axis. Default 0
  sweepAngle?: Num;    // degrees; positive = CCW. Default 360
}

interface Polygon extends Base, Stroke { type: "polygon"; width?: Size; height?: Size; polygonCount?: Num; cornerRadius?: Num; fill?: Fills; effect?: Effect | Effect[] }

interface Path extends Base, Stroke {
  type: "path"; width?: Size; height?: Size; fill?: Fills; effect?: Effect | Effect[];
  geometry?: string;                         // SVG path data
  viewBox?: [number, number, number, number];   // region of path space stretched onto the node box. Default: tight bounds
  fillRule?: "nonzero" | "evenodd";
}

interface TextStyle {
  fontFamily?: string | Var;   // any Google Font
  fontSize?: Num;
  fontWeight?: string | Var;   // "400", "600", "normal", "bold", ...
  fontStyle?: string | Var;    // "italic"
  letterSpacing?: Num;         // px
  lineHeight?: Num;            // multiple of fontSize. Default: the font's own line height
  textAlign?: "left" | "center" | "right" | "justify";
  textAlignVertical?: "top" | "middle" | "bottom";   // only with fixed-width-height
  underline?: Bool; strikethrough?: Bool;
  href?: string;
}

interface Text extends Base, Stroke, TextStyle {
  type: "text";
  content?: string | Var;
  fill?: Fills;               // text is invisible without a fill
  effect?: Effect | Effect[];
  /** "auto" (default): one line, size from the content; width/height ignored.
   *  "fixed-width": wraps at `width`; height from the content.
   *  "fixed-width-height": wraps at `width`, fixed `height` (content may overflow). */
  textGrowth?: "auto" | "fixed-width" | "fixed-width-height";
  width?: Size; height?: Size;
}

interface Icon extends Base {
  type: "icon";
  library?: "lucide" | "feather" | "phosphor" | "Material Symbols Outlined" | "Material Symbols Rounded" | "Material Symbols Sharp";
  icon?: string;          // icon name in that library, e.g. "search"
  weight?: Num;           // Material Symbols only, 100–700
  width?: Size; height?: Size;   // the icon scales to fit
  fill?: Fills;           // icon color
  effect?: Effect | Effect[];
}

/** An instance of a reusable node. Any of the component root's properties can be set on the ref
 *  to override them (e.g. width, fill). Descendants are customized through `descendants`. */
interface Ref extends Base {
  type: "ref";
  ref: string;   // id of the reusable node
  /** Keys are descendant ids (or "nestedInstanceId/childId" inside nested instances).
   *  Without `type`: property overrides. With `type`: replaces that descendant with a new subtree.
   *  `children: [...]` (without `type`) replaces that descendant's children. */
  descendants?: Record<string, object>;
  [property: string]: any;
}

interface Note extends Base, TextStyle { type: "note" | "prompt" | "context"; content?: string | Var; width?: Size; height?: Size; model?: string }
interface Script extends Base { type: "script"; scriptUri?: string; inputs?: Record<string, any>; width?: Size; height?: Size; clip?: Bool }

type Node = Frame | Group | Rectangle | Ellipse | Polygon | Path | Text | Icon | Ref | Note | Script;

/** Variables. Themed values pick the entry whose `theme` matches the node's active theme
 *  (document defaults are the first value of each theme axis). */
type Variables = Record<string,
  { type: "color" | "number" | "string" | "boolean"; value: any | { value: any; theme?: Record<string, string> }[] }>;
type Themes = Record<string, string[]>;   // axis -> values, e.g. { mode: ["light", "dark"] }
```
